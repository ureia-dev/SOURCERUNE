cmake_minimum_required(VERSION 3.25)
project(SOURCERUNE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

add_library(sourcerune_dsp STATIC Source/DSP/SceneProcessor.cpp)
target_include_directories(sourcerune_dsp PUBLIC Source/DSP)
if(NOT MSVC)
    target_compile_options(sourcerune_dsp PRIVATE -ffp-contract=off)
endif()

option(SOURCERUNE_BUILD_VST3 "Build the native VST3 MVP" ON)
option(SOURCERUNE_BUILD_HOST_SMOKE "Build the manual, bounded SDK host smoke" OFF)
set(SOURCERUNE_VST3_SDK "" CACHE PATH "Existing VST3 SDK checkout (with submodules)")
if(SOURCERUNE_BUILD_VST3)
    # No VSTGUI/editor rewrite, DAW installation, or automatic full validator run.
    set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
    set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "Explicit manual opt-in full validator only")
    if(NOT SOURCERUNE_VST3_SDK)
        include(FetchContent)
        FetchContent_Declare(vst3sdk
            GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git
            GIT_TAG 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96
            GIT_SUBMODULES base cmake pluginterfaces public.sdk
        )
        FetchContent_MakeAvailable(vst3sdk)
        set(SOURCERUNE_VST3_SDK "${vst3sdk_SOURCE_DIR}")
    else()
        add_subdirectory("${SOURCERUNE_VST3_SDK}" "${CMAKE_BINARY_DIR}/vst3sdk")
    endif()
    smtg_enable_vst3_sdk()
    smtg_add_vst3plugin(SOURCERUNE Source/Plugin/VST3/Plugin.cpp)
    target_link_libraries(SOURCERUNE PRIVATE sourcerune_dsp sdk)
    target_include_directories(SOURCERUNE PRIVATE "${SOURCERUNE_VST3_SDK}")
    if(APPLE)
        smtg_target_set_bundle(SOURCERUNE BUNDLE_IDENTIFIER "dev.ureia.sourcerune" COMPANY_NAME "SOURCERUNE")
    endif()
    smtg_target_add_plugin_resources(SOURCERUNE RESOURCES "${SOURCERUNE_VST3_SDK}/LICENSE.txt")

    if(SOURCERUNE_BUILD_HOST_SMOKE)
        if(APPLE)
            enable_language(OBJCXX)
            set(module_source "${SOURCERUNE_VST3_SDK}/public.sdk/source/vst/hosting/module_mac.mm")
            set_source_files_properties("${module_source}" PROPERTIES COMPILE_FLAGS "-fobjc-arc")
        elseif(WIN32)
            set(module_source "${SOURCERUNE_VST3_SDK}/public.sdk/source/vst/hosting/module_win32.cpp")
        else()
            set(module_source "${SOURCERUNE_VST3_SDK}/public.sdk/source/vst/hosting/module_linux.cpp")
        endif()
        add_executable(sourcerune_host_smoke Tests/Audio/vst3_host_smoke.cpp
            "${module_source}" "${SOURCERUNE_VST3_SDK}/public.sdk/source/common/memorystream.cpp")
        target_include_directories(sourcerune_host_smoke PRIVATE "${SOURCERUNE_VST3_SDK}")
        target_link_libraries(sourcerune_host_smoke PRIVATE sdk_hosting sourcerune_dsp ${CMAKE_DL_LIBS})
        if(APPLE)
            find_library(COCOA_FRAMEWORK Cocoa REQUIRED)
            target_link_libraries(sourcerune_host_smoke PRIVATE "${COCOA_FRAMEWORK}")
        endif()
    endif()
endif()
